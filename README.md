        function applySearchFilter(keyword, cat) {
            const cards = document.querySelectorAll('[data-product-card]');
            cards.forEach(card => {
                const pid = card.getAttribute('data-product-card');
                const p = allProductsMap[pid];
                if (!p) return;
                const matchKeyword = !keyword || p.name.toLowerCase().includes(keyword) || p.category.toLowerCase().includes(keyword);
                const matchCat = (cat === 'all') || (p.catKey === cat);
                card.style.display = (matchKeyword && matchCat) ? 'flex' : 'none';
            });
        }

        /* ============================================================================
           5. GIỎ HÀNG & FORM YÊU CẦU BÁO GIÁ
        ============================================================================ */
        let cart = [];

        function addToCart(productId) {
            const p = allProductsMap[productId];
            if (!p) return;
            const existing = cart.find(item => item.id === productId);
            if (existing) {
                existing.qty += 1;
            } else {
                cart.push({ id: p.id, name: p.name, price: p.price, qty: 1 });
            }
            updateCartUI();
            openCartModal();
        }

        function removeFromCart(productId) {
            cart = cart.filter(item => item.id !== productId);
            updateCartUI();
        }

        function updateCartUI() {
            const totalQty = cart.reduce((sum, i) => sum + i.qty, 0);
            document.getElementById('cartCountBadge').textContent = totalQty;

            const container = document.getElementById('cartItemsContainer');
            if (cart.length === 0) {
                container.innerHTML = `<div class="py-6 text-center text-slate-500">Chưa có sản phẩm trong giỏ hàng.</div>`;
                document.getElementById('cartTotalPrice').textContent = '0 ₫';
                return;
            }

            let total = 0;
            container.innerHTML = cart.map(item => {
                const sub = item.price * item.qty;
                total += sub;
                return `
                <div class="py-2.5 flex items-center justify-between gap-2">
                    <div class="pr-2">
                        <div class="font-bold text-slate-800">${item.name}</div>
                        <div class="text-slate-500">${formatVND(item.price)} x ${item.qty}</div>
                    </div>
                    <div class="flex items-center gap-2 shrink-0">
                        <span class="font-bold text-nqred-600">${formatVND(sub)}</span>
                        <button onclick="removeFromCart('${item.id}')" class="w-6 h-6 rounded bg-rose-50 text-rose-600 hover:bg-rose-100 flex items-center justify-center">
                            <i class="fa-solid fa-trash-can text-[11px]"></i>
                        </button>
                    </div>
                </div>`;
            }).join('');
            document.getElementById('cartTotalPrice').textContent = formatVND(total);
        }

        function openCartModal() {
            updateCartUI();
            const modal = document.getElementById('cartModal');
            modal.classList.remove('hidden');

        }
